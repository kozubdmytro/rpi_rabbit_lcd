cmake_minimum_required(VERSION 3.16)
project(smartrabbit-test LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Threads REQUIRED)
add_library(sr_core src/framebuffer.cpp src/renderer.cpp)
target_include_directories(sr_core PUBLIC src)
target_compile_options(sr_core PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(sr_core PUBLIC Threads::Threads)
add_executable(smartrabbit-test src/main.cpp src/lcd_renderer.cpp)
target_link_libraries(smartrabbit-test PRIVATE rt m)
find_package(PkgConfig REQUIRED)
pkg_check_modules(GIO REQUIRED IMPORTED_TARGET gio-2.0)
add_library(sr_gateway src/mesh/gateway.cpp src/mesh/bluez.cpp src/screens/readings.cpp src/web.cpp)
target_include_directories(sr_gateway PUBLIC src)
target_compile_options(sr_gateway PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(sr_gateway PUBLIC sr_core PkgConfig::GIO)
target_link_libraries(smartrabbit-test PRIVATE sr_gateway)
enable_testing()
add_executable(smartrabbit-tests tests/test.cpp src/screens/readings.cpp)
target_link_libraries(smartrabbit-tests PRIVATE sr_core)
add_test(NAME framebuffer_and_rotation COMMAND smartrabbit-tests)
add_executable(smartrabbit-network-tests tests/network_test.cpp src/screens/readings.cpp)
target_link_libraries(smartrabbit-network-tests PRIVATE sr_core)
add_test(NAME wifi_and_node_removal COMMAND smartrabbit-network-tests)

add_executable(smartrabbit-menu-tests tests/gateway_menu_test.cpp)
target_link_libraries(smartrabbit-menu-tests PRIVATE sr_gateway)
add_test(NAME menu_and_reset_retry COMMAND smartrabbit-menu-tests)
